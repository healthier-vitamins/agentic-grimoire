% \recipe{name}{the formula or numbered moves, then one short example}: a box written out in
% full inside the step that uses it, so the reader never scrolls back for it. A shortcut that
% does several steps at once goes after them: \recipe{Name: Steps 1 to 4 in one go}{...}.
\newcommand{\recipe}[2]{\par\smallskip\noindent\fcolorbox{curve}{curve!6}{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule}{\color{black}\textbf{\faReceipt\ Recipe --- memorise this: #1}\par #2}}\par\smallskip}
% First half of a solution: recipes, then \fast{} (the recipe run on this question in 2-4 lines), then \ans{}.
\newcommand{\fast}[1]{\par\smallskip\textbf{Fast route.}\par #1}
% Second half: the full steps, for when there is time. Step numbers restart at 1.
\newcommand{\whyworks}{\par\bigskip\setcounter{stepno}{0}\textbf{Why it works} {\small\textit{(skip if rushing, come back later)}}\par}
% \choice: a fine-print note at the start of a step that needs a decision.
\newcommand{\choice}{{\footnotesize\textit{(a choice, not a fixed recipe move)}}\ }
% \reason{}: the move in words (what cancels, where it goes), with a bar on the left, before the full equation.
\newcommand{\reason}[1]{\par\smallskip\noindent{\color{curve}\vrule width 2pt}\hspace{6pt}\parbox{\dimexpr\linewidth-8pt}{\color{black}#1}\par}
% \side{}: a small side sum worked out before the full equation, in a grey box.
\newcommand{\side}[1]{\par\smallskip\noindent\colorbox{gray!15}{\small\color{black}side sum: #1}\par}
% \note{}: the 1-4 word reason for one line of an align* block, in its own right-hand column.
